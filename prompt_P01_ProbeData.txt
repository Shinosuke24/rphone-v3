package com.rphone.v3.model

/**
 * Data hasil pengukuran Probe Meter dari ESP32.
 *
 * Format JSON masuk dari ESP32:
 *   GET_VOLT  : {"probe":"VOLT","volt":3.280,"display":"3.280V"}
 *   GET_DIODE : {"probe":"DIODE","vdrop":0.648,"display":"648mV"}
 *               {"probe":"OPEN","display":"OL"}
 *               {"probe":"SHORT","display":"0mV"}
 *   GET_OHM   : {"probe":"OHM","ohm":4700.0,"display":"4.70KΩ"}
 *               {"probe":"OPEN","display":"OL"}
 *               {"probe":"SHORT","display":"0Ω"}
 *
 * Field mode mengikuti nilai "probe" dari JSON:
 *   "VOLT", "DIODE", "OHM", "OPEN", "SHORT"
 */
data class ProbeData(
    val mode: String = "",        // "VOLT" | "DIODE" | "OHM" | "OPEN" | "SHORT"
    val volt: Float = 0f,         // diisi saat mode VOLT (dalam Volt)
    val vdrop: Float = 0f,        // diisi saat mode DIODE (dalam Volt, misal 0.648)
    val ohm: Float = 0f,          // diisi saat mode OHM (dalam Ohm)
    val display: String = ""      // string siap tampil dari ESP32, misal "3.280V", "648mV", "4.70KΩ", "OL", "0Ω"
)
